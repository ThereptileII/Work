cmake_minimum_required(VERSION 3.24)
project(OpenNavCommissioningRestartNative LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This standalone gate requires native Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /W4)
endif()
get_filename_component(OPENNAV_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(restart_guard
  ${OPENNAV_SOURCE}/src/platform/windows/CommissioningRestartProtocol.cpp
  ${OPENNAV_SOURCE}/src/platform/windows/CommissioningRestartNative.cpp)
target_include_directories(restart_guard PUBLIC ${OPENNAV_SOURCE}/src)
target_link_libraries(restart_guard PUBLIC bcrypt advapi32)
add_executable(opennav-restart ${OPENNAV_SOURCE}/src/platform/windows/RestartHelper.cpp)
target_link_libraries(opennav-restart PRIVATE restart_guard)
add_executable(opencpn marker_process.cpp
  ${OPENNAV_SOURCE}/src/platform/windows/WindowsPlatformIntegration.cpp)
target_link_libraries(opencpn PRIVATE restart_guard)
add_executable(commissioning_restart_tests ${OPENNAV_SOURCE}/tests/commissioning_restart_tests.cpp)
target_link_libraries(commissioning_restart_tests PRIVATE restart_guard)
enable_testing()
add_test(NAME portable_wire_contract COMMAND commissioning_restart_tests)
