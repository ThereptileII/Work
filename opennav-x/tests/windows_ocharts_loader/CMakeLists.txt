cmake_minimum_required(VERSION 3.24)
project(skager_windows_ocharts_loader LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Native MSVC Win32 required")
endif()
get_filename_component(OPENNAV_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(wxWidgets_USE_REL_AND_DBG OFF)
find_package(wxWidgets REQUIRED COMPONENTS base core)
include(${wxWidgets_USE_FILE})
if(NOT EXISTS "${PICOSHA2_DIR}/picosha2.h")
  message(FATAL_ERROR "Exact pinned picosha2 header required")
endif()
add_executable(ocharts_loader_test
  loader_test.cpp
  "${OPENNAV_ROOT}/src/integration/OChartsModuleLoader.cpp"
  "${OPENNAV_ROOT}/src/integration/PluginPresentationLoader.cpp")
target_include_directories(ocharts_loader_test PRIVATE "${OPENNAV_ROOT}/src" "${PICOSHA2_DIR}")
target_link_libraries(ocharts_loader_test PRIVATE ${wxWidgets_LIBRARIES})
add_executable(win32_pe_forwarders_test ../win32_pe_forwarders_test.cpp)
target_include_directories(win32_pe_forwarders_test PRIVATE "${OPENNAV_ROOT}/src")
add_executable(plugin_tls_byte_policy_test ../plugin_tls_byte_policy_test.cpp)
target_include_directories(plugin_tls_byte_policy_test PRIVATE
  "${OPENNAV_ROOT}/src" "${PICOSHA2_DIR}")
target_compile_features(plugin_tls_byte_policy_test PRIVATE cxx_std_17)
target_compile_options(plugin_tls_byte_policy_test PRIVATE /utf-8 /W4)
set_property(TARGET plugin_tls_byte_policy_test PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_library(legacy_dependency SHARED legacy_dependency.cpp)
set_target_properties(legacy_dependency PROPERTIES OUTPUT_NAME libeay32)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/legacy-ordinal.def" "EXPORTS\n  legacy_tls_fixture @209\n")
target_sources(legacy_dependency PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/legacy-ordinal.def")
add_library(ssleay_dependency SHARED legacy_dependency.cpp "${CMAKE_CURRENT_BINARY_DIR}/legacy-ordinal.def")
set_target_properties(ssleay_dependency PROPERTIES OUTPUT_NAME ssleay32
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/forwarders")
# MSVC emits ordinary local aliases. The runner constructs the exact forwarder
# in this bounded export-name reservation, retaining all code/import bytes.
string(REPEAT "X" 96 forwarder_padding)
set(forwarder_slot "__skager_forwarder_slot_${forwarder_padding}")
foreach(kind IN ITEMS clean first dotted helper)
  add_library(forward_${kind} SHARED forward_dependency.cpp)
  set_target_properties(forward_${kind} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/forwarders")
  set(exports "EXPORTS\n  fixture_value @209\n")
  if(kind STREQUAL "clean")
    string(APPEND exports "  dll.fixture_value=fixture_other_value\n")
  elseif(kind STREQUAL "first")
    set_target_properties(forward_first PROPERTIES OUTPUT_NAME forward)
    target_compile_definitions(forward_first PRIVATE FORWARD_VALUE=309)
    string(APPEND exports "  clean.fixture_value=fixture_other_value\n")
  elseif(kind STREQUAL "dotted")
    set_target_properties(forward_dotted PROPERTIES OUTPUT_NAME forward.clean)
    target_compile_definitions(forward_dotted PRIVATE FORWARD_VALUE=409)
  elseif(kind STREQUAL "helper")
    string(APPEND exports "  forwarded_probe=fixture_value\n  ${forwarder_slot}=fixture_value\n")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/forward_${kind}.def" "${exports}")
  target_sources(forward_${kind} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/forward_${kind}.def")
endforeach()
add_library(helper_legacy SHARED indirect_dependency.cpp)
target_compile_definitions(helper_legacy PRIVATE INDIRECT_LEGACY)
target_link_libraries(helper_legacy PRIVATE legacy_dependency)
add_library(helper_clean SHARED indirect_dependency.cpp)
add_library(helper_unmapped SHARED unmapped_dependency.cpp)
# Keep this unique helper out of the executable/application directory.
set_target_properties(helper_unmapped PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/unmapped")
foreach(variant IN ITEMS good no_bind no_status no_create no_destroy legacy_normal legacy_delay indirect_legacy indirect_clean indirect_unmapped forward_legacy_name forward_legacy_ordinal forward_ssleay_ordinal forward_indirect forward_clean_name forward_clean_ordinal forward_explicit_dll forward_multi_dot)
  add_library(fixture_${variant} SHARED fixture.cpp)
  target_include_directories(fixture_${variant} PRIVATE "${OPENNAV_ROOT}/src")
  if(variant MATCHES "^legacy_")
    target_compile_definitions(fixture_${variant} PRIVATE LEGACY_IMPORT)
    target_link_libraries(fixture_${variant} PRIVATE legacy_dependency)
    if(variant STREQUAL "legacy_delay")
      target_link_libraries(fixture_${variant} PRIVATE delayimp)
      target_link_options(fixture_${variant} PRIVATE /DELAYLOAD:libeay32.dll)
    endif()
  endif()
  if(variant MATCHES "^indirect_")
    target_compile_definitions(fixture_${variant} PRIVATE INDIRECT_IMPORT)
    string(REPLACE "indirect_" "" helper_kind "${variant}")
    target_link_libraries(fixture_${variant} PRIVATE helper_${helper_kind})
  endif()
  set(exports "EXPORTS\n")
  foreach(entry IN ITEMS bind status create destroy)
    if(variant STREQUAL "no_${entry}")
      string(TOUPPER "OMIT_${entry}" omit)
      target_compile_definitions(fixture_${variant} PRIVATE "${omit}")
    else()
      if(entry STREQUAL "bind")
        string(APPEND exports "  skager_bind_chart_presentation_v1\n")
      elseif(entry STREQUAL "status")
        string(APPEND exports "  skager_chart_presentation_status_v1\n")
      else()
        string(APPEND exports "  ${entry}_pi\n")
      endif()
    endif()
  endforeach()
  if(variant MATCHES "^forward_")
    set_target_properties(fixture_${variant} PROPERTIES
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/forwarders")
    target_compile_definitions(fixture_${variant} PRIVATE FORWARDER_FIXTURE)
    string(APPEND exports "  forwarded_probe=fixture_forwarder_placeholder\n  ${forwarder_slot}=fixture_forwarder_placeholder\n")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${variant}.def" "${exports}")
  target_sources(fixture_${variant} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${variant}.def")
endforeach()
foreach(target IN ITEMS ocharts_loader_test win32_pe_forwarders_test fixture_good fixture_no_bind fixture_no_status fixture_no_create fixture_no_destroy fixture_legacy_normal fixture_legacy_delay fixture_indirect_legacy fixture_indirect_clean fixture_indirect_unmapped legacy_dependency helper_legacy helper_clean helper_unmapped ssleay_dependency forward_clean forward_first forward_dotted forward_helper fixture_forward_legacy_name fixture_forward_legacy_ordinal fixture_forward_ssleay_ordinal fixture_forward_indirect fixture_forward_clean_name fixture_forward_clean_ordinal fixture_forward_explicit_dll fixture_forward_multi_dot)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE /utf-8 /W4)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endforeach()
