cmake_minimum_required(VERSION 3.24)
project(FrameRecaptureWindows LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This focus regression requires native Windows")
endif()
find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
include(${wxWidgets_USE_FILE})
if(NOT IS_DIRECTORY "${RECAPTURE_EXTRACTED}/guarded" OR
   NOT IS_DIRECTORY "${RECAPTURE_EXTRACTED}/original")
  message(FATAL_ERROR "Authenticated production method extraction required")
endif()
add_library(recapture_controls STATIC ../../src/ui/Controls.cpp ../../src/vessel/VesselState.cpp)
target_include_directories(recapture_controls PUBLIC ../../src)
target_compile_features(recapture_controls PUBLIC cxx_std_17)
target_compile_definitions(recapture_controls PUBLIC XNAV_ENABLE_TEST_FIXTURES=0)
target_compile_options(recapture_controls PRIVATE /utf-8 /W4)
target_link_libraries(recapture_controls PUBLIC ${wxWidgets_LIBRARIES} dwrite)
foreach(variant guarded original)
  add_executable(recapture_${variant} main.cpp)
  target_include_directories(recapture_${variant} PRIVATE "${RECAPTURE_EXTRACTED}/${variant}")
  target_compile_definitions(recapture_${variant} PRIVATE OPENNAV_X)
  if(variant STREQUAL "guarded")
    target_compile_definitions(recapture_${variant} PRIVATE EXPECTED_SETUP_GUARD=1)
  else()
    target_compile_definitions(recapture_${variant} PRIVATE EXPECTED_SETUP_GUARD=0)
  endif()
  target_compile_options(recapture_${variant} PRIVATE /utf-8 /W4)
  target_link_libraries(recapture_${variant} PRIVATE recapture_controls)
endforeach()
