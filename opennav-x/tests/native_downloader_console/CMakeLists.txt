cmake_minimum_required(VERSION 3.24)
project(skager_native_downloader_console LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Native MSVC Win32 required")
endif()
set(wxWidgets_USE_REL_AND_DBG OFF)
find_package(wxWidgets REQUIRED COMPONENTS base core)
include(${wxWidgets_USE_FILE})
add_executable(native-downloader-console probe.cpp)
target_include_directories(native-downloader-console PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../tools")
target_compile_features(native-downloader-console PRIVATE cxx_std_17)
target_compile_options(native-downloader-console PRIVATE /utf-8)
target_compile_definitions(native-downloader-console PRIVATE wxDEBUG_LEVEL=1)
set_property(TARGET native-downloader-console PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_link_libraries(native-downloader-console PRIVATE ${wxWidgets_LIBRARIES})
