cmake_minimum_required(VERSION 3.24)
project(skager_chart_binding_runtime LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Native MSVC Win32 required")
endif()
foreach(mode IN ITEMS original guarded)
  add_executable(binding-${mode} probe.cpp)
  target_include_directories(binding-${mode} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../src")
  target_compile_features(binding-${mode} PRIVATE cxx_std_17)
  target_compile_options(binding-${mode} PRIVATE /utf-8 /W4)
  set_property(TARGET binding-${mode} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_link_libraries(binding-${mode} PRIVATE version)
endforeach()
# Match pinned OpenCPN CMakeLists.txt; this is the sole A/B compile difference.
target_compile_definitions(binding-guarded PRIVATE _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
