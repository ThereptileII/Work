cmake_minimum_required(VERSION 3.20)
project(observable_concurrency LANGUAGES CXX)
if(NOT EXISTS "${OPENCPN_SOURCE_DIR}/libs/observable/src/observable.cpp")
  message(FATAL_ERROR "OPENCPN_SOURCE_DIR must name the staged pinned source")
endif()
find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
find_package(Threads REQUIRED)
include(${wxWidgets_USE_FILE})
add_executable(observable_concurrency main.cpp
  "${OPENCPN_SOURCE_DIR}/libs/observable/src/observable.cpp")
target_include_directories(observable_concurrency PRIVATE
  "${OPENCPN_SOURCE_DIR}/libs/observable/include")
target_compile_features(observable_concurrency PRIVATE cxx_std_17)
target_link_libraries(observable_concurrency PRIVATE ${wxWidgets_LIBRARIES} Threads::Threads)
option(OBSERVABLE_SANITIZE "Enable AddressSanitizer and UBSan (GCC/Clang)" OFF)
if(OBSERVABLE_SANITIZE)
  if(MSVC)
    message(FATAL_ERROR "Use the ordinary native MSVC fixture on Windows")
  endif()
  target_compile_options(observable_concurrency PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(observable_concurrency PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
foreach(scenario IN ITEMS unregister register move stress)
  add_test(NAME observable_${scenario} COMMAND observable_concurrency ${scenario})
  set_tests_properties(observable_${scenario} PROPERTIES TIMEOUT 15)
endforeach()
