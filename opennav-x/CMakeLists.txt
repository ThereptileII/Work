cmake_minimum_required(VERSION 3.24)
project(OpenNavX VERSION 0.1.0 LANGUAGES CXX)
if(MSVC)
  # Labels and paths are UTF-8 regardless of the Windows runner's code page.
  add_compile_options(/utf-8)
endif()

option(OPENNAV_BUILD_TESTS "Build portable OpenNav contract tests" ON)
option(OPENNAV_BUILD_UI_COMPONENTS "Build wxWidgets OpenNav UI primitives" OFF)
if(WIN32)
  add_library(opennav_platform src/platform/windows/WindowsPlatformIntegration.cpp)
  add_executable(opennav-restart src/platform/windows/RestartHelper.cpp)
  target_include_directories(opennav-restart PRIVATE src)
  target_compile_features(opennav-restart PRIVATE cxx_std_17)
else()
  add_library(opennav_platform src/platform/linux/LinuxPlatformIntegration.cpp)
endif()
target_include_directories(opennav_platform PUBLIC src)
target_compile_features(opennav_platform PUBLIC cxx_std_17)
target_sources(opennav_platform PRIVATE src/platform/PortableProfile.cpp)
# Upstream is built separately and stays pristine until baseline gates pass.
add_library(opennav_integration src/integration/StartupMode.cpp src/integration/StartupRecovery.cpp src/integration/RouteProgressInput.cpp src/integration/PreviewResources.cpp)
add_library(opennav_vessel src/vessel/VesselState.cpp src/vessel/NavigationInput.cpp src/vessel/RouteProgress.cpp src/vessel/DemoSource.cpp src/vessel/SensorRegistry.cpp)
target_link_libraries(opennav_integration PUBLIC opennav_vessel)
target_include_directories(opennav_vessel PUBLIC src)
target_compile_features(opennav_vessel PUBLIC cxx_std_17)
add_library(opennav_smartnav src/smartnav/Energy.cpp src/smartnav/VesselEnergy.cpp src/smartnav/EnergyConfiguration.cpp src/smartnav/Advisories.cpp src/smartnav/HazardLookAhead.cpp)
target_include_directories(opennav_smartnav PUBLIC src)
target_link_libraries(opennav_smartnav PUBLIC opennav_vessel)
target_compile_features(opennav_smartnav PUBLIC cxx_std_17)
add_library(opennav_application src/application/Settings.cpp src/application/MarineMapping.cpp)
target_include_directories(opennav_application PUBLIC src)
target_link_libraries(opennav_application PUBLIC opennav_smartnav)
target_compile_features(opennav_application PUBLIC cxx_std_17)
add_library(opennav_adapters src/adapters/Autopilot.cpp)
target_include_directories(opennav_adapters PUBLIC src)
target_link_libraries(opennav_adapters PUBLIC opennav_vessel)
target_compile_features(opennav_adapters PUBLIC cxx_std_17)
target_include_directories(opennav_integration PUBLIC src)
target_compile_features(opennav_integration PUBLIC cxx_std_17)
set_target_properties(opennav_integration PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(opennav_integration PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(opennav_integration PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(OPENNAV_BUILD_UI_COMPONENTS)
  find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base aui)
  include(${wxWidgets_USE_FILE})
  add_library(opennav_ui src/ui/Controls.cpp src/ui/Shell.cpp src/ui/PreviewPanel.cpp src/ui/ProductPanel.cpp src/ui/ProductSettings.cpp src/ui/Sheet.cpp)
  target_include_directories(opennav_ui PUBLIC src)
  target_link_libraries(opennav_ui PUBLIC opennav_vessel opennav_smartnav opennav_adapters opennav_application ${wxWidgets_LIBRARIES})
  target_compile_features(opennav_ui PUBLIC cxx_std_17)
endif()

  add_test(NAME sensor_battery_coherence COMMAND sensor_registry_tests battery)

  add_test(NAME autopilot_manual_feedback COMMAND adapter_tests manual)
  add_test(NAME autopilot_failures COMMAND adapter_tests failure)

  add_executable(settings_tests tests/settings_tests.cpp)
  target_link_libraries(settings_tests PRIVATE opennav_application)
  add_test(NAME settings_record_roundtrip COMMAND settings_tests roundtrip)

  add_test(NAME settings_explicit_propulsion_mappings COMMAND settings_tests mappings)
  add_executable(demo_tests tests/demo_tests.cpp)
  target_link_libraries(demo_tests PRIVATE opennav_smartnav)

  target_link_libraries(windows_arguments_tests PRIVATE opennav_platform)
