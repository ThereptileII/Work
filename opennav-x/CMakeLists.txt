cmake_minimum_required(VERSION 3.24)
project(OpenNavX VERSION 0.1.0 LANGUAGES CXX)
if(MSVC)
  # Labels and paths are UTF-8 regardless of the Windows runner's code page.
  add_compile_options(/utf-8)
endif()

option(OPENNAV_BUILD_TESTS "Build portable OpenNav contract tests" ON)
option(OPENNAV_BUILD_UI_COMPONENTS "Build wxWidgets OpenNav UI primitives" OFF)
if(WIN32)
  add_library(opennav_platform src/platform/windows/WindowsPlatformIntegration.cpp)
  add_executable(opennav-restart src/platform/windows/RestartHelper.cpp)
  target_include_directories(opennav-restart PRIVATE src)
  target_compile_features(opennav-restart PRIVATE cxx_std_17)
else()
  add_library(opennav_platform src/platform/linux/LinuxPlatformIntegration.cpp)
endif()
target_include_directories(opennav_platform PUBLIC src)
target_compile_features(opennav_platform PUBLIC cxx_std_17)
# Upstream is built separately and stays pristine until baseline gates pass.
add_library(opennav_integration src/integration/StartupMode.cpp)
add_library(opennav_vessel src/vessel/VesselState.cpp)
target_include_directories(opennav_vessel PUBLIC src)
target_compile_features(opennav_vessel PUBLIC cxx_std_17)
target_include_directories(opennav_integration PUBLIC src)
target_compile_features(opennav_integration PUBLIC cxx_std_17)
set_target_properties(opennav_integration PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(opennav_integration PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(opennav_integration PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(OPENNAV_BUILD_UI_COMPONENTS)
  find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base aui)
  include(${wxWidgets_USE_FILE})
  add_library(opennav_ui src/ui/Controls.cpp src/ui/Shell.cpp)
  target_include_directories(opennav_ui PUBLIC src)
  target_link_libraries(opennav_ui PUBLIC opennav_vessel ${wxWidgets_LIBRARIES})
  target_compile_features(opennav_ui PUBLIC cxx_std_17)
endif()

if(OPENNAV_BUILD_TESTS)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(restart_probe tests/restart_probe.cpp)
  target_link_libraries(restart_probe PRIVATE opennav_platform)
  if(WIN32)
    add_dependencies(restart_probe opennav-restart)
  endif()
  add_test(NAME restart_lifecycle_contract COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/restart_lifecycle.py $<TARGET_FILE:restart_probe>)
  set_tests_properties(restart_lifecycle_contract PROPERTIES TIMEOUT 30)
  add_executable(startup_mode_tests tests/startup_mode_tests.cpp)
  target_link_libraries(startup_mode_tests PRIVATE opennav_integration)
  add_test(NAME startup_mode_contract COMMAND startup_mode_tests)
  add_executable(vessel_state_tests tests/vessel_state_tests.cpp)
  target_link_libraries(vessel_state_tests PRIVATE opennav_vessel)
  add_test(NAME vessel_quality_contract COMMAND vessel_state_tests)
  add_executable(windows_arguments_tests tests/windows_arguments_tests.cpp)
  target_link_libraries(windows_arguments_tests PRIVATE opennav_platform)
  if(WIN32)
    target_link_libraries(windows_arguments_tests PRIVATE shell32)
  endif()
  add_test(NAME restart_arguments_contract COMMAND windows_arguments_tests)
endif()
